  window.addEventListener('load', () => setTimeout(() => pre.classList.add('is-gone'), 280));

  // Hero slider
  const slides = [...document.querySelectorAll('.hero-slide')];
  const dotsWrap = document.getElementById('heroDots');
  const currentEl = document.getElementById('heroCurrent');
  let heroIndex = 0, timer;
  slides.forEach((_,i) => {
    const b = document.createElement('button');
    b.className = 'hero-dot' + (i===0?' is-active':'');
    b.setAttribute('aria-label','Go to slide '+(i+1));
    b.addEventListener('click',()=>go(i,true));
    dotsWrap.appendChild(b);
  });
  const dots = [...dotsWrap.children];
  function go(i,manual=false){
    slides[heroIndex].classList.remove('is-active'); dots[heroIndex].classList.remove('is-active');
    heroIndex = (i + slides.length) % slides.length;
    slides[heroIndex].classList.add('is-active'); dots[heroIndex].classList.add('is-active');
    currentEl.textContent = String(heroIndex+1).padStart(2,'0');
    if(manual) restart();
  }
  function restart(){ clearInterval(timer); timer=setInterval(()=>go(heroIndex+1),6500); }
  document.getElementById('heroNext').addEventListener('click',()=>go(heroIndex+1,true));
  document.getElementById('heroPrev').addEventListener('click',()=>go(heroIndex-1,true));
  restart();

  // Mobile menu
  const panel=document.getElementById('mobilePanel'), scrim=document.getElementById('mobileScrim'), menuBtn=document.getElementById('menuBtn');
  const closeMenu=()=>{panel.classList.remove('is-open');scrim.classList.remove('is-open')};
  menuBtn.addEventListener('click',()=>{panel.classList.toggle('is-open');scrim.classList.toggle('is-open')});
  scrim.addEventListener('click',closeMenu);
  panel.querySelectorAll('a').forEach(a=>a.addEventListener('click',closeMenu));

  // Reveal
  const io = new IntersectionObserver(entries=>entries.forEach(e=>{if(e.isIntersecting){e.target.classList.add('is-visible');io.unobserve(e.target)}}),{threshold:.12});
  document.querySelectorAll('.reveal').forEach(el=>io.observe(el));

  // Booking modal
  const modal=document.getElementById('bookingModal');
  const openModal=()=>{modal.classList.add('is-open');document.body.classList.add('modal-open')};
  const closeModal=()=>{modal.classList.remove('is-open');document.body.classList.remove('modal-open')};
  document.querySelectorAll('[data-open-booking]').forEach(b=>b.addEventListener('click',()=>{closeMenu();openModal()}));
  document.querySelectorAll('[data-close-booking]').forEach(b=>b.addEventListener('click',closeModal));

  // Demo forms
  const toast=document.getElementById('toast');
  function showToast(text='Demo request captured — no message was sent.'){toast.textContent=text;toast.classList.add('show');setTimeout(()=>toast.classList.remove('show'),3200)}
  document.querySelectorAll('.demo-form').forEach(form=>form.addEventListener('submit',e=>{e.preventDefault();showToast();if(form.id==='bookingForm') closeModal();}));

  // Gallery filters
  const filterBtns=[...document.querySelectorAll('.filter-btn')], tiles=[...document.querySelectorAll('.gallery-tile')];
  filterBtns.forEach(btn=>btn.addEventListener('click',()=>{
    filterBtns.forEach(b=>b.classList.remove('is-active'));btn.classList.add('is-active');
    const f=btn.dataset.filter;
    tiles.forEach(t=>t.classList.toggle('is-hidden',f!=='all' && t.dataset.cat!==f));
  }));

  // Lightbox
  const lb=document.getElementById('lightbox'), lbImg=document.getElementById('lightboxImage');
  tiles.forEach(t=>t.addEventListener('click',()=>{lbImg.src=t.dataset.lightbox;lb.classList.add('is-open');document.body.classList.add('modal-open')}));
  const closeLb=()=>{lb.classList.remove('is-open');document.body.classList.remove('modal-open')};
  document.getElementById('lightboxClose').addEventListener('click',closeLb);
  lb.addEventListener('click',e=>{if(e.target===lb) closeLb()});

  // FAQ assistant
  const chatBox=document.getElementById('chatBox'), chatLog=document.getElementById('chatLog'), chatInput=document.getElementById('chatInput');
  document.getElementById('chatLaunch').addEventListener('click',()=>chatBox.classList.toggle('is-open'));
  document.getElementById('chatClose').addEventListener('click',()=>chatBox.classList.remove('is-open'));
  const replies={
    rooms:"Sea Breeze offers hotel accommodation in a relaxed coastal setting. Specific room categories, rates and occupancy details can be added once the hotel confirms them.",
    weddings:"Yes. Sea Breeze lists three banquet venues — White Horse, White Elephant and White Lion — and also has attractive beach and garden settings for celebrations.",
    dining:"The hotel includes a seafood restaurant, and the property materials show Sri Lankan dining and beachside dining settings.",
    location:"Sea Breeze Hotel is in the Ambalangoda area on Sri Lanka’s southern coast, with a beachfront setting.",
    services:"The hotel lists 24-hour room service, parking and 24-hour security. The property also offers gardens, a pool, dining and event spaces.",
    price:"Room rates and event package prices are not included in this demo yet. Those can be added after the hotel confirms the current pricing.",
    contact:"Contact details have been intentionally left out of this concept until the hotel confirms the preferred telephone, email and enquiry channels.",
    default:"I can help with general information about rooms, weddings and events, dining, location, or hotel services. For rates and specific availability, the final site will connect to the hotel’s confirmed enquiry channel."
  };
  function addBubble(text,type){const div=document.createElement('div');div.className='bubble '+type;div.textContent=text;chatLog.appendChild(div);chatLog.scrollTop=chatLog.scrollHeight}
  function answer(q){
    const s=q.toLowerCase();
    let key='default';
    if(/room|stay|accommodation|bed/.test(s)) key='rooms';
    else if(/wedding|event|banquet|party|birthday|hall/.test(s)) key='weddings';
    else if(/food|dining|restaurant|seafood|breakfast/.test(s)) key='dining';
    else if(/where|location|address|ambalangoda|map/.test(s)) key='location';
    else if(/service|parking|security|pool|garden/.test(s)) key='services';
    else if(/rate|price|cost|charge|package/.test(s)) key='price';
    else if(/phone|email|contact|whatsapp/.test(s)) key='contact';
    setTimeout(()=>addBubble(replies[key],'bot'),240);
  }
  document.querySelectorAll('[data-q]').forEach(b=>b.addEventListener('click',()=>{addBubble(b.textContent,'user');answer(b.dataset.q)}));
  document.getElementById('chatForm').addEventListener('submit',e=>{e.preventDefault();const q=chatInput.value.trim();if(!q)return;addBubble(q,'user');chatInput.value='';answer(q)});

  // ESC handling
  document.addEventListener('keydown',e=>{if(e.key==='Escape'){closeLb();closeModal();closeMenu();chatBox.classList.remove('is-open')}});
  document.getElementById('year').textContent=new Date().getFullYear();
})();
